cmake_minimum_required(VERSION 3.21)
project(wasm_allocator_workers LANGUAGES C CXX)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "This regression exercises Emscripten startup and shared memory")
endif()
add_executable(allocator-workers main.cpp ../../src/wasm/host_allocator.cpp ../../src/wasm/host_allocator_wasm.cpp ../../src/wasm/host_mspace.c)
target_include_directories(allocator-workers PRIVATE ../../src)
target_compile_features(allocator-workers PRIVATE cxx_std_17)
target_compile_options(allocator-workers PRIVATE -pthread)
target_compile_definitions(allocator-workers PRIVATE NK_WASM_HOST_HEAP_LIMIT=16777216)
target_link_options(allocator-workers PRIVATE -pthread -sPTHREAD_POOL_SIZE=4 -sWASM_WORKERS=1 -sMALLOC=none -sINITIAL_MEMORY=33554432 -sALLOW_MEMORY_GROWTH=0 -sASSERTIONS=1)
